\begin{frame}{1. Diagnóstico y Contexto del Sistema Penitenciario}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Población Penal y Proyección}
        \begin{itemize}
          \item \textbf{162.043 personas} atendidas en el sistema penal chileno (88,2\% hombres / 11,8\% mujeres).
          \item \textbf{30.000 egresos} proyectados entre 2025 y 2030; \textbf{8,2\% son mujeres}.
          \item \textbf{85,4\% de las mujeres} son madres y jefas de hogar sustentadoras.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Eficiencia Fiscal y Hacinamiento}
        \begin{itemize}
          \item Hacinamiento crítico en régimen cerrado: promedio \textbf{130\% de ocupación} (hasta 200\% en alta densidad).
          \item Costo penal cerrado: \textbf{\$800.000 CLP/mes} por persona recluida.
          \item Medio libre / intermediación: inversión \textbf{5 veces menor} por usuario con alto impacto protector.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% LÁMINA 3: PROBLEMATIZACIÓN Y SESGOS B2B
% -----------------------------------------------------------------------------
\begin{frame}{2. Problematización y Desafío B2B}
  \begin{itemize}
    \item \textbf{Sesgo de Deseabilidad Social Declarativa:}
      \begin{itemize}
        \item Las encuestas directas a gerentes ocultan el \textbf{veto implícito} en selección.
        \item Respuestas guiadas por corrección política vs.\ decisiones reales de contratación.
      \end{itemize}
    \vspace{0.2cm}
    \item \textbf{Doble Penalización de Género:}
      \begin{itemize}
        \item Estigma social por antecedentes penales + sanción por quebrantar el rol tradicional de género.
        \item Altas cargas no remuneradas de cuidado familiar que colisionan con jornadas rígidas.
      \end{itemize}
    \vspace{0.2cm}
    \item \textbf{Brecha de Inseguridad vs.\ Victimización:}
      \begin{itemize}
        \item Victimización real en caída (23,5\%), pero percepción de inseguridad en máximo histórico (56,6\%).
        \item Amplifica la resistencia infundada en las gerencias de gestión de personas.
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{3. Preguntas y Objetivos Estratégicos}
  \begin{block}{Pregunta Central}
    ¿Cuáles son los mecanismos organizacionales, condiciones necesarias y sesgos de género que explican la adopción empresarial y la permanencia laboral de mujeres en la red de JxR?
  \end{block}

  \vspace{0.2cm}

  \begin{block}{Objetivos Específicos}
    \begin{enumerate}
      \item \textbf{Línea Base:} Medir permanencia observada a 30, 60 y 90 días por género.
      \item \textbf{Aislar Sesgo:} Medir discriminación implícita mediante viñetas pareadas.
      \item \textbf{Modelamiento QCA:} Combinatoria de condiciones para la adopción B2B.
      \item \textbf{Propuesta Operativa:} Rediseñar oferta JxR (Bolsa Laboral, OTEC y CET).
    \end{enumerate}
  \end{block}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{6. Modelo QCA: Parsimonia y Condiciones Causales}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{block}{3 Condiciones Organizacionales ($k=3$)}
        \begin{enumerate}
          \item \textbf{Sponsor en la Dirección:} Liderazgo de alta gerencia alineado.
          \item \textbf{Compatibilidad de Cuidados:} Flexibilidad ante cargas familiares.
          \item \textbf{Acompañamiento Post-Colocación:} Soporte psicosocial continuo de JxR.
        \end{enumerate}
      \end{block}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{block}{Justificación Metodológica}
        \begin{itemize}
          \item $k=3 \implies 2^3 = 8$ combinaciones configuracionales.
          \item Con $N=10$ casos, la razón $10/8 > 1$ evita la \textit{diversidad limitada} (celdas vacías).
          \item Maximiza la validez interna y la solidez lógica.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

    \item \textbf{Comparación Emparejada (6 Casos de Contratación):}
      \begin{itemize}
        \item 3 Casos con permanencia continua vs.\ 3 casos con interrupción laboral.
        \item Pareados por rubro económico y tamaño de empresa.
      \end{itemize}
    \vspace{0.2cm}
    \item \textbf{Díadas Espejadas (Jefatura -- Trabajadora):}
      \begin{itemize}
        \item Entrevistas en paralelo y contrastadas en 3 casos empíricos.
        \item Permite reconstruir la experiencia cotidiana en el puesto de trabajo.
      \end{itemize}
    \vspace{0.2cm}
    \item \textbf{Foco Analítico:}
      \begin{itemize}
        \item Identificar detonantes de quiebre vs.\ factores de arraigo laboral.
        \item Evaluar el rol del soporte temprano durante los primeros 90 días.
      \end{itemize}
  \end{itemize}
\end{frame}
